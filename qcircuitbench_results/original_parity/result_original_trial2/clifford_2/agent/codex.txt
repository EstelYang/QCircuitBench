import numpy as np
from itertools import product

H = np.array([[1,1],[1,-1]],dtype=complex)/np.sqrt(2)
S = np.array([[1,0],[0,1j]],dtype=complex)
I = np.eye(2,dtype=complex)

# brute search sequences up to length 12 over {H,S}
# and also allow repeating S multiple times; but that's included.
ops={'h':H,'s':S}

def is_global_phase(U, tol=1e-9):
 # check if U is scalar*I
 if np.allclose(U,0):
  return False,None
 # pick element (0,0)
 a=U[0,0]
 if abs(a)<tol:
  a=U[0,1]
 # scalar guess
 g=a
 if np.allclose(U,g*I,atol=tol):
  return True,g
 return False,None

for L in range(1,13):
 for seq in product('hs', repeat=L):
  U=I
  for ch in seq:
   U=ops[ch]@U
  ok,g=is_global_phase(U)
  if ok:
   # normalize phase
   g/=abs(g)
   # check if g is ω or ω"'^k
   # print if g close to exp(i*pi/4)
   target=np.exp(1j*np.pi/4)
   if abs(g-target)<1e-9:
    print('"'found', ''.join(seq), 'len',L,'phase',g)
    raise SystemExit
print('none')
