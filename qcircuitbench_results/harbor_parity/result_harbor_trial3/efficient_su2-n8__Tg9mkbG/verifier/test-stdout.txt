from qiskit_algorithms.optimizers import COBYLA
from qiskit.primitives import Estimator


def run_and_analyze(circuit, hamiltonian, expected_energy, optiter=200, tolerance=1e-3):
    optimizer = COBYLA(maxiter=optiter)
    estimator = Estimator()
    vqe = VQE(estimator, circuit, optimizer)

    result = vqe.compute_minimum_eigenvalue(hamiltonian)
    energy = result.eigenvalue
    if hasattr(energy, "real"):
        energy = energy.real

    error = abs(float(energy) - float(expected_energy))
    return error
"""
