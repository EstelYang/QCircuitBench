code_string = """from qiskit import transpile

def run_and_analyze(circuit, aer_sim):
    n = 8

    def rref_rank(rows):
        rows = rows[:]
        r = 0
        pivot_cols = []
        for col in range(n):
            pivot = None
            for i in range(r, len(rows)):
                if (rows[i] >> col) & 1:
                    pivot = i
                    break
            if pivot is None:
                continue
            rows[r], rows[pivot] = rows[pivot], rows[r]
            for i in range(len(rows)):
                if i != r and ((rows[i] >> col) & 1):
                    rows[i] ^= rows[r]
            pivot_cols.append(col)
            r += 1
            if r == n:
                break
        return rows, r, pivot_cols

    def nullspace_vector(rows_rref, pivot_cols):
        free_cols = [c for c in range(n) if c not in pivot_cols]
        if not free_cols:
            return 0
        if len(free_cols) != 1:
            return None

        free = free_cols[0]
        x = 1 << free
        for row in rows_rref:
            pivot_col = None
            for c in pivot_cols:
                if (row >> c) & 1:
                    pivot_col = c
                    break
            if pivot_col is None:
                continue
            parity_other = (row & x).bit_count() & 1
            if parity_other:
                x |= 1 << pivot_col
        return x

    transpiled = transpile(circuit, aer_sim, optimization_level=0)
    unique_rows = set()

    shots_per_round = 4096
    max_rounds = 16
    for _ in range(max_rounds):
        result = aer_sim.run(transpiled, shots=shots_per_round).result()
        counts = result.get_counts(transpiled)
        for bitstring in counts:
            y = int(bitstring, 2)
            if y != 0:
                unique_rows.add(y)

        rows_rref, rank, pivot_cols = rref_rank(list(unique_rows))
        if rank >= n - 1:
            candidate = nullspace_vector(rows_rref, pivot_cols)
            if candidate is not None and candidate != 0:
                prediction = format(candidate, "08b")
                return prediction

    rows_rref, rank, pivot_cols = rref_rank(list(unique_rows))
    candidate = nullspace_vector(rows_rref, pivot_cols)
    if candidate is None or candidate == 0:
        for s_candidate in range(1, 1 << n):
            ok = True
            for y in unique_rows:
                if ((y & s_candidate).bit_count() & 1) != 0:
                    ok = False
                    break
            if ok:
                prediction = format(s_candidate, "08b")
                return prediction

    prediction = format(candidate, "08b")
    return prediction
"""
