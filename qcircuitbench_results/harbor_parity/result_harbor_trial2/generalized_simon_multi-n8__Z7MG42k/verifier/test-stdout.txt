code_string = """from qiskit import transpile

def _score_candidates(counts, n):
    total = sum(counts.values())
    scores = [0.0] * (1 << n)
    for s in range(1, 1 << n):
        orth = 0
        for bitstr, cnt in counts.items():
            y = int(bitstr, 2)
            if ((y & s).bit_count() & 1) == 0:
                orth += cnt
        scores[s] = orth / total
    return scores

def run_and_analyze(circuit, aer_sim):
    n = 8
    shots = 2048
    best1 = 1
    best2 = 2

    for _ in range(3):
        compiled = transpile(circuit, aer_sim, optimization_level=0)
        result = aer_sim.run(compiled, shots=shots).result()
        counts = result.get_counts()

        scores = _score_candidates(counts, n)
        ranked = sorted(range(1, 1 << n), key=lambda s: scores[s], reverse=True)
        best1, best2 = ranked[0], ranked[1]
        if scores[best1] >= 0.65 and scores[best2] >= 0.65 and best1 != best2:
            break
        shots *= 2

    prediction1 = format(best1, "0{}b".format(n))
    prediction2 = format(best2, "0{}b".format(n))
    if prediction2 < prediction1:
        prediction1, prediction2 = prediction2, prediction1
    return prediction1, prediction2
"""
