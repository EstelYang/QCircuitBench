code_string = """from typing import Optional

from qiskit import transpile


def _parity(x: int) -> int:
    return bin(x).count("1") & 1


def _find_s_from_equations(equations: set[int], n: int) -> Optional[int]:
    if not equations:
        return None
    candidates = []
    for s in range(1, 1 << n):
        ok = True
        for y in equations:
            if _parity(y & s) != 0:
                ok = False
                break
        if ok:
            candidates.append(s)
            if len(candidates) > 1:
                break
    return candidates[0] if len(candidates) == 1 else None


def run_and_analyze(circuit, aer_sim):
    n = 8
    equations = set()
    shots = 4096
    tqc = transpile(circuit, aer_sim)

    while shots <= 131072:
        result = aer_sim.run(tqc, shots=shots).result()
        counts = result.get_counts()

        for bitstring in counts:
            y = int(bitstring, 2)
            if y != 0:
                equations.add(y)

        s_int = _find_s_from_equations(equations, n)
        if s_int is not None:
            prediction = format(s_int, f"0{n}b")
            return prediction

        shots *= 2

    prediction = "0" * n
    return prediction
"""
